function convertIPAtoViet(ipa) { if (!ipa) return ""; // Làm sạch IPA: Xóa dấu gạch chéo, dấu chấm ngắt âm tiết, dấu ngoặc let text = ipa.toLowerCase().replace(/[\/\.\(\)ˌ]/g, '').trim(); // 1. Đánh dấu trọng âm (dấu sắc) text = text.replace(/ˈ([^aeiouăơêôiy]*)([aeiouæɑɒɔəʊʌiuv])/, 'ˈ$1$2́'); text = text.replace(/ˈ/g, ''); // 2. Bảng ánh xạ âm phức tạp (Thứ tự ưu tiên từ dài đến ngắn) const complexRules = [ { r: /tʃ/g, v: 'ch' }, { r: /dʒ/g, v: 'd' }, { r: /aɪ/g, v: 'ai' }, { r: /eɪ/g, v: 'êi' }, { r: /ɔɪ/g, v: 'oi' }, { r: /aʊ/g, v: 'ao' }, { r: /əʊ/g, v: 'âu' }, { r: /ɪə/g, v: 'ia' }, { r: /eə/g, v: 'e-ơ' }, { r: /ʊə/g, v: 'ua' }, { r: /θ/g, v: 'th' }, { r: /ð/g, v: 'đ' }, { r: /ʃ/g, v: 'sh' }, { r: /ʒ/g, v: 'd' }, { r: /ŋ/g, v: 'ng' } ]; complexRules.forEach(rule => { text = text.replace(rule.r, rule.v); }); // 3. Bảng ánh xạ âm đơn const simpleRules = [ { r: /æ|ɑː|ɒ|ɔː/g, v: 'o' }, { r: /ʌ/g, v: 'ă' }, { r: /ə/g, v: 'ơ' }, { r: /ɛ|e/g, v: 'e' }, { r: /ɪ|iː/g, v: 'i' }, { r: /ʊ|uː/g, v: 'u' }, { r: /f/g, v: 'ph' }, { r: /s/g, v: 'x' }, { r: /z/g, v: 'z' }, { r: /j/g, v: 'i' }, { r: /w/g, v: 'u' }, { r: /ɹ|r/g, v: 'r' } ]; simpleRules.forEach(rule => { text = text.replace(rule.r, rule.v); }); // 4. Xử lý phụ âm cuối để đọc đúng kiểu bồi (Ví dụ: t -> tờ) text = text.replace(/t$/g, ' tờ').replace(/k$/g, ' cờ') .replace(/p$/g, ' pờ').replace(/s$/g, ' xờ') .replace(/d$/g, ' dờ').replace(/g$/g, ' gờ'); return text.toUpperCase(); }
